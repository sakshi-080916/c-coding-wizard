Question 5. Swap numbers using Call by address .

Program:

#include <stdio.h>
#include <conio.h>
void swap(int *x, int *y)
{
int z;
z = *x;
*x = *y;
*y = z;
}


void main()
{
int a, b;
clrscr();   
printf("Enter two numbers: ");
scanf("%d %d", &a, &b);
printf("\n a = %d\n b = %d", a, b);
swap(&a, &b);  
printf("swapping:\na = %d\nb = %d", a, b);
}

#include <conio.h>

{

Question 1: Create structure of employee which there is emp.ID, name, designation of 50 employees from user, print details in tabular form.

Program:

#include <stdio.h>
#include <conio.h>

struct details
{
    int id;
    char name[40], des[40];
};
void main()
{
    int i, j;
    struct details emp[50];
    clrscr();
    printf("Enter the data of employees : \n");
    for (i = 0; i < 50; i++)
    {
        printf("Emp %d\n", i+1 );
        printf("Name: ");
        gets(emp[i].name);
	fflush(stdin);
        printf("\nEnter the designation: ");
        gets(emp[i].des);
        printf("\nEnter Emp ID: ");
        scanf("%d", &emp[i].id);
        
    }
    printf("--------------------------------------------------------------------------------");
    printf("| %-25s | %-25s | %-35s |", "ID", "Designation","Name");
    for (i = 0; i <50; i++)
    {
	printf("--------------------------------------------------------------------------------");
        printf("| %-25d | %-25s | %-35s |", emp[i].id, emp[i].des,emp[i].name);
    }
 
}

Question 3: Create structure of employee which there is emp.ID, name, designation of 50 employees from user and print name character with emp.ID in tabular form.

#include <stdio.h>
#include <conio.h>

struct details
{
    int id;
    char name[35], des[40];
};

void main()
{
    int i, j;
    struct details emp[50];
    clrscr();
    printf("Enter the data for the employees below: \n");
    for (i = 0; i < 50; i++)
    {
        printf("Emp %d\n", i + 1);
        printf("Name: ");
        gets(emp[i].name);
        fflush(stdin);
        printf("\nEnter the designation: ");
        gets(emp[i].des);
        printf("\nEnter the Employee ID: ");
        scanf("%d", &emp[i].id);

    printf("--------------------------------------------------------------------------");

    for (i = 0; i < 50; i++)
    {
        printf("-----------------------------------------------------------------------");
        printf("| %-20d | %-45s |", emp[i].id, emp[i].name);
    }
}

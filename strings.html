Question 13: String palindrome example (using reverse).

Program:

#include <string.h>
#include<conio.h>

void main()
{
char str1[50],str2[50];
int i,len;
clrscr();
printf("Enter a word: ");
gets(str1);
len=strlen(str1);
for(i=0;str1[i]!='\0';i++)
{
str2[i]=str1[len-1-i];
}
str2[i]=0;
l=strcmp(str1,str2);
if(l==0)printf("That is a pallindrome");
else
 printf("That is not a pallindrome");

}
